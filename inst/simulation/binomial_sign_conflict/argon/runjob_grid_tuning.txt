##   qsub -N binom_sim_final  runjob_grid_tuning.txt final_30reps.R 1:9 1:30
##   qsub -N binom_sim_final2 runjob_grid_tuning.txt final_30reps.R "c(11,12,13,15,16,19,22)" 1:30
